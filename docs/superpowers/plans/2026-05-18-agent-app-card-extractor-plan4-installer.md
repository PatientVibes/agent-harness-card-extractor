# agent-app-card-extractor — Plan 4 of 4: Installer

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Package the FastAPI + Vite app as a single-file PyInstaller executable that launches uvicorn on a free localhost port, opens the user's default browser to the UI, and lets them drop a PDF + run the chat flow end-to-end — no Python install required.

**Architecture:** Two changes to the running app and three new files for packaging. (1) FastAPI gains a static-asset mount that serves the Vite `web/dist/` build for non-API routes (with SPA fallback). (2) `AppConfig` learns to resolve resource paths (static dir) from the PyInstaller bundle (`sys._MEIPASS`) and place writable state under a per-user data directory (`platformdirs.user_data_dir`) when frozen. (3) A new `app/launcher.py` picks a free port, starts uvicorn in-process, waits for `/health` to respond, and opens the default browser. (4) A new `packaging/pyinstaller.spec` and `packaging/build.py` drive the bundle. The harness's runtime resources (prompts, schema files) are pulled in via `--collect-all card_extractor`.

**Tech Stack:** PyInstaller 6.x, `platformdirs` (cross-platform user data dirs), stdlib `socket` + `webbrowser` + `threading`, existing FastAPI/uvicorn. No frontend changes — Plan 3b's bundle is consumed as-is.

**Scope boundary:** No auto-update mechanism. No installer chrome (no MSI, no .pkg, no signed binary) — `dist/agent-app-card-extractor.exe` is the deliverable. No Linux-specific support beyond best-effort (PyInstaller produces a Linux binary if built on Linux; not tested in CI). Frontend code is untouched — Plan 3b's `web/dist/` is the artifact. This plan ends when (a) `python -m app.launcher` opens the UI and processes a PDF locally and (b) the PyInstaller binary in `dist/` does the same on a clean shell with no `uv`/`pip` available.

```python
"""FastAPI app entry — lifespan owns the durable state."""
from __future__ import annotations

import os
import sqlite3
from contextlib import asynccontextmanager
from pathlib import Path

from fastapi import FastAPI
from fastapi.staticfiles import StaticFiles

from card_extractor.agent import DurableContext  # type: ignore

from app.config import AppConfig
from app.db import init_db
from app.routes import chat, health, jobs, uploads
from app.services.pipeline import PipelineService


def _web_dist_dir(cfg: AppConfig) -> Path | None:
    """Resolve the directory of the built frontend, or None if not present."""
    candidate = getattr(cfg, "web_dist", None)
    if candidate is None:
        candidate_env = os.environ.get("APP_WEB_DIST")
        candidate = Path(candidate_env) if candidate_env else None
    if candidate is None:
        return None
    return candidate if candidate.is_dir() else None


def build_app() -> FastAPI:
    @asynccontextmanager
    async def lifespan(app: FastAPI):
        cfg = AppConfig.from_env()
        cfg.ensure_dirs()
        init_db(cfg.db_path)
        conn = sqlite3.connect(cfg.db_path, check_same_thread=False)
        conn.row_factory = sqlite3.Row

        os.environ.setdefault("WIKI_DIR", str(cfg.state_dir / "knowledge"))
        os.environ.setdefault("REVIEW_QUEUE_DIR", str(cfg.state_dir / "review_queue"))

        durable = DurableContext.from_env()
        durable.output_dir = cfg.state_dir / "harness_out"
        durable.output_dir.mkdir(parents=True, exist_ok=True)

        app.state.config = cfg
        app.state.db = conn
        app.state.durable = durable
        app.state.pipeline = PipelineService(durable=durable, config=cfg)
        app.state.background_tasks = set()
        try:
            yield
        finally:
            conn.close()

    app = FastAPI(title="agent-app-card-extractor", version="0.1.0", lifespan=lifespan)
    app.include_router(health.router)
    app.include_router(uploads.router)
    app.include_router(jobs.router)
    app.include_router(chat.router)

    cfg_for_static = AppConfig.from_env()
    web_dist = _web_dist_dir(cfg_for_static)
    if web_dist is not None:
        app.mount("/", StaticFiles(directory=str(web_dist), html=True), name="static")
    return app


app = build_app()
```

```python
"""AppConfig — environment-driven paths for state, uploads, outputs, and web assets."""
from __future__ import annotations

import os
import sys
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional

from platformdirs import user_data_dir


_APP_NAME = "agent-app-card-extractor"
_APP_AUTHOR = "card-extractor"


def _is_frozen() -> bool:
    return getattr(sys, "frozen", False) is True


def _bundle_root() -> Optional[Path]:
    """PyInstaller's extraction root (--onefile) or executable dir (--onedir)."""
    meipass = getattr(sys, "_MEIPASS", None)
    if meipass:
        return Path(str(meipass))
    return None


def _default_state_dir() -> Path:
    env = os.environ.get("APP_STATE_DIR")
    if env:
        return Path(env)
    if _is_frozen():
        return Path(user_data_dir(_APP_NAME, _APP_AUTHOR))
    # Source layout default: <repo>/app/state/
    return Path(__file__).resolve().parent / "state"


def _default_web_dist() -> Optional[Path]:
    env = os.environ.get("APP_WEB_DIST")
    if env:
        return Path(env)
    if _is_frozen():
        bundle = _bundle_root()
        if bundle is not None:
            candidate = bundle / "web_dist"
            return candidate if candidate.is_dir() else None
        return None
    # Source layout: <repo>/web/dist/
    candidate = Path(__file__).resolve().parents[1] / "web" / "dist"
    return candidate if candidate.is_dir() else None


@dataclass
class AppConfig:
    state_dir: Path
    web_dist: Optional[Path] = None

    @classmethod
    def from_env(cls) -> "AppConfig":
        return cls(state_dir=_default_state_dir(), web_dist=_default_web_dist())

    @property
    def db_path(self) -> Path:
        return self.state_dir / "jobs.db"

    @property
    def uploads_dir(self) -> Path:
        return self.state_dir / "uploads"

    @property
    def outputs_dir(self) -> Path:
        return self.state_dir / "outputs"

    def ensure_dirs(self) -> None:
        self.state_dir.mkdir(parents=True, exist_ok=True)
        self.uploads_dir.mkdir(parents=True, exist_ok=True)
        self.outputs_dir.mkdir(parents=True, exist_ok=True)
```

The launcher is the entry point inside the PyInstaller bundle. It:
1. Picks a free TCP port on `127.0.0.1` by binding to port 0.
2. Builds a uvicorn `Server` programmatically (so we can poll `server.started` from a worker thread).
3. Starts the server on a background thread.
4. Waits up to 10s for `server.started` to flip.
5. Opens `webbrowser.open_new_tab(f"http://127.0.0.1:{port}/")`.
6. Joins the server thread — exits cleanly on Ctrl-C.

We split the testable bits (`find_free_port`, `wait_for_server`) from the entry point (`main`) which we don't test end-to-end here (Task 5's smoke does that).

```python
"""Launcher — picks a port, starts uvicorn, opens the browser.

This is the entry point invoked by the PyInstaller binary. It can also be run
during development as `python -m app.launcher` to exercise the same flow
(useful for testing the static-asset mount end-to-end without the dev proxy).
"""
from __future__ import annotations

import logging
import socket
import sys
import threading
import time
import webbrowser
from typing import Any

import uvicorn

logger = logging.getLogger("app.launcher")


def find_free_port() -> int:
    """Return an OS-assigned free port on 127.0.0.1."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind(("127.0.0.1", 0))
        return int(s.getsockname()[1])


def wait_for_server(server: Any, timeout_seconds: float = 10.0) -> bool:
    """Poll the uvicorn Server's `started` flag with a 50ms tick. Returns True
    if the server reported started before the timeout, False otherwise."""
    deadline = time.monotonic() + timeout_seconds
    while time.monotonic() < deadline:
        if getattr(server, "started", False):
            return True
        time.sleep(0.05)
    return False


def open_browser(port: int) -> None:
    webbrowser.open_new_tab(f"http://127.0.0.1:{port}/")


def main() -> int:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(name)s: %(message)s")
    port = find_free_port()
    logger.info("Selected port %d", port)

    # Lazy import so the test suite doesn't have to import FastAPI at module load.
    from app.main import build_app

    config = uvicorn.Config(
        app=build_app(),
        host="127.0.0.1",
        port=port,
        log_level="info",
        # Reload off — this is the bundled entry point, not the dev server.
        reload=False,
    )
    server = uvicorn.Server(config)

    server_thread = threading.Thread(target=server.run, name="uvicorn", daemon=True)
    server_thread.start()

    if not wait_for_server(server, timeout_seconds=10.0):
        logger.error("uvicorn did not report started within 10s — aborting")
        server.should_exit = True
        server_thread.join(timeout=5.0)
        return 1

    open_browser(port)
    logger.info("Browser opened to http://127.0.0.1:%d/", port)

    try:
        server_thread.join()
    except KeyboardInterrupt:
        logger.info("Shutdown requested")
        server.should_exit = True
        server_thread.join(timeout=5.0)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# -*- mode: python ; coding: utf-8 -*-
"""PyInstaller spec for agent-app-card-extractor.

Entry point: app/launcher.py
Output: dist/agent-app-card-extractor(.exe) — single-file binary.

The spec must be invoked from the repo root:

    pyinstaller packaging/pyinstaller.spec --noconfirm
"""
from pathlib import Path
from PyInstaller.utils.hooks import collect_all, collect_data_files

REPO_ROOT = Path.cwd()

# --- Harness package: gather everything (modules + datas + binaries) ---
harness_datas, harness_binaries, harness_hiddenimports = collect_all("card_extractor")

# --- Bundled frontend ---
web_dist = REPO_ROOT / "web" / "dist"
if not web_dist.is_dir():
    raise RuntimeError(
        f"web/dist is not built. Run `npm --prefix web run build` before pyinstaller."
    )
# PyInstaller does NOT accept glob patterns in `datas` — expand to a list of
# (source_file, dest_dir_inside_bundle) tuples.
web_datas: list[tuple[str, str]] = []
for path in web_dist.rglob("*"):
    if path.is_file():
        rel_parent = path.relative_to(web_dist).parent
        dest = str(Path("web_dist") / rel_parent) if str(rel_parent) != "." else "web_dist"
        web_datas.append((str(path), dest))

# --- uvicorn hidden imports (dynamic loaders) ---
uvicorn_hiddenimports = [
    "uvicorn.lifespan.on",
    "uvicorn.lifespan.off",
    "uvicorn.protocols.http.auto",
    "uvicorn.protocols.http.h11_impl",
    "uvicorn.protocols.websockets.auto",
    "uvicorn.protocols.websockets.websockets_impl",
    "uvicorn.protocols.websockets.wsproto_impl",
    "uvicorn.loops.auto",
    "uvicorn.loops.asyncio",
    "uvicorn.logging",
]

a = Analysis(
    [str(REPO_ROOT / "app" / "launcher.py")],
    pathex=[str(REPO_ROOT)],
    binaries=harness_binaries,
    datas=harness_datas + web_datas,
    hiddenimports=harness_hiddenimports + uvicorn_hiddenimports,
    hookspath=[],
    runtime_hooks=[],
    excludes=["tkinter", "test", "tests"],
    noarchive=False,
)

pyz = PYZ(a.pure, a.zipped_data)

exe = EXE(
    pyz,
    a.scripts,
    a.binaries,
    a.zipfiles,
    a.datas,
    [],
    name="agent-app-card-extractor",
    debug=False,
    bootloader_ignore_signals=False,
    strip=False,
    upx=False,
    upx_exclude=[],
    runtime_tmpdir=None,
    console=True,
    disable_windowed_traceback=False,
    argv_emulation=False,
    target_arch=None,
    codesign_identity=None,
    entitlements_file=None,
)
```

```python
"""Build the agent-app-card-extractor PyInstaller binary.

Steps:
  1. Build the Vite frontend → web/dist/.
  2. Run PyInstaller against packaging/pyinstaller.spec.

Usage (from repo root):
    uv run python packaging/build.py
"""
from __future__ import annotations

import shutil
import subprocess
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[1]


def _run(cmd: list[str], cwd: Path | None = None) -> None:
    print(f"$ {' '.join(cmd)}", flush=True)
    result = subprocess.run(cmd, cwd=str(cwd) if cwd else None)
    if result.returncode != 0:
        sys.exit(result.returncode)


def main() -> int:
    web_dir = REPO_ROOT / "web"
    if not (web_dir / "node_modules").is_dir():
        _run(["npm", "install"], cwd=web_dir)
    _run(["npm", "run", "build"], cwd=web_dir)

    spec = REPO_ROOT / "packaging" / "pyinstaller.spec"
    _run(["pyinstaller", str(spec), "--noconfirm", "--clean"], cwd=REPO_ROOT)

    dist = REPO_ROOT / "dist"
    print(f"\nBuild complete. Binary in: {dist}")
    for entry in sorted(dist.iterdir()):
        size_mb = entry.stat().st_size / (1024 * 1024)
        print(f"  {entry.name}  ({size_mb:.1f} MB)")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### Task 5: Binary smoke test

**Files:**
- Create: `D:\agent-app-card-extractor\tests\test_packaging_smoke.py`

A single integration test that:
1. Builds the binary if `dist/agent-app-card-extractor[.exe]` is absent (skips on CI if no `npm` / network).
2. Launches it as a subprocess.
3. Polls `http://127.0.0.1:<scanned-port>/health` to detect the listening port (the binary picks a random one — we scan the recent ports in the launcher log instead).
4. Asserts `{"status":"ok"}` comes back.
5. Kills the binary.

Marking with `@pytest.mark.integration` and `@pytest.mark.skipif(...)` so it doesn't run in default `pytest` and doesn't fail CI when the binary doesn't exist.

The trickiest part is **knowing what port the binary picked.** Three options:
- (a) Read the binary's stdout (it logs `Selected port NNNN`).
- (b) Pre-build the binary with `APP_PORT` env override (would require launcher changes — out of scope for this task).
- (c) Scan ports.

Going with (a) — line-buffered subprocess + parse stdout.

- [ ] **Step 1: Write the test**

`D:\agent-app-card-extractor\tests\test_packaging_smoke.py`:

```python
"""Integration smoke test for the PyInstaller binary.

Run with: `uv run pytest -m integration tests/test_packaging_smoke.py -v`
Skipped by default — set RUN_PACKAGING_SMOKE=1 to enable.
"""
from __future__ import annotations

import os
import re
import subprocess
import sys
import time
from pathlib import Path

import httpx
import pytest

REPO_ROOT = Path(__file__).resolve().parents[1]
BIN_PATH = REPO_ROOT / "dist" / ("agent-app-card-extractor.exe" if sys.platform == "win32" else "agent-app-card-extractor")
PORT_RE = re.compile(r"Selected port (\d+)")


pytestmark = pytest.mark.integration


@pytest.mark.skipif(
    os.environ.get("RUN_PACKAGING_SMOKE") != "1",
    reason="set RUN_PACKAGING_SMOKE=1 to run the binary smoke test",
)
@pytest.mark.skipif(
    not BIN_PATH.exists(),
    reason=f"{BIN_PATH} not found — run `uv run python packaging/build.py` first",
)
def test_binary_serves_health():
    proc = subprocess.Popen(
        [str(BIN_PATH)],
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
        bufsize=1,
    )
    try:
        port = _read_port_from_stdout(proc, timeout_seconds=30.0)
        assert port is not None, "Binary did not report 'Selected port N' within 30s"

        # Hit /health
        for attempt in range(50):
            try:
                r = httpx.get(f"http://127.0.0.1:{port}/health", timeout=2.0)
                if r.status_code == 200:
                    body = r.json()
                    assert body.get("status") == "ok"
                    return  # success
            except httpx.HTTPError:
                pass
            time.sleep(0.2)
        pytest.fail(f"/health never returned 200 on port {port}")
    finally:
        proc.terminate()
        try:
            proc.wait(timeout=5.0)
        except subprocess.TimeoutExpired:
            proc.kill()


def _read_port_from_stdout(proc: subprocess.Popen, timeout_seconds: float) -> int | None:
    deadline = time.monotonic() + timeout_seconds
    while time.monotonic() < deadline:
        line = proc.stdout.readline() if proc.stdout else ""
        if line:
            print(f"[binary] {line.rstrip()}")
            m = PORT_RE.search(line)
            if m:
                return int(m.group(1))
        if proc.poll() is not None:
            return None
    return None
```
